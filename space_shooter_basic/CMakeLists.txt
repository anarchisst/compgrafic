cmake_minimum_required(VERSION 3.20)
project(space_defender_basic LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw URL https://codeload.github.com/glfw/glfw/zip/refs/tags/3.4
    URL_HASH SHA256=a133ddc3d3c66143eba9035621db8e0bcf34dba1ee9514a9e23e96afd39fd57a)
FetchContent_Declare(glm URL https://codeload.github.com/g-truc/glm/zip/refs/tags/1.0.1
    URL_HASH SHA256=09c5716296787e1f7fcb87b1cbdbf26814ec1288ed6259ccd30d5d9795809fa5)
FetchContent_MakeAvailable(glfw glm)
find_package(OpenGL REQUIRED)
add_executable(space_defender_basic main.cpp ../external/glad/src/gl.c)
target_include_directories(space_defender_basic PRIVATE ../external/glad/include)
target_compile_definitions(space_defender_basic PRIVATE GLFW_INCLUDE_NONE)
target_link_libraries(space_defender_basic PRIVATE glfw glm::glm OpenGL::GL)
if(MSVC)
    target_compile_options(space_defender_basic PRIVATE /W4)
else()
    target_compile_options(space_defender_basic PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
endif()
if(MINGW)
    target_link_options(space_defender_basic PRIVATE -static-libgcc -static-libstdc++)
endif()
